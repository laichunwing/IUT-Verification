# 24_corollary_3_12_numerical.sage
#
# Unit X of "Independent Exact Verification of the Finite-Sum and
# Scalar-Rearrangement Identities Surrounding IUT IV".
#
# Verifies the numerical content of IUT III Corollary 3.12 for
# 11a1 at p=11, l=13.
#
# Classification: Exact / Direct (symbolic in log(11)).
# Runs in SageMath 9.5+ or 10.x. Calls main() unconditionally.

from sage.all import QQ, log, N

def main():
    print("=" * 78)
    print("  UNIT X: NUMERICAL CONTENT OF COROLLARY 3.12 FOR 11a1")
    print("=" * 78)
    print()

    p = 11
    ell = 13
    ell_star = (ell - 1) // 2
    m = 5  # v_11(q_11)

    # Symbolic: log(11) is exact.
    L_q = m * log(p)   # symbolic
    L_q_float = float(L_q)  # for display only

    print(f"  Curve: 11a1, p = {p}, l = {ell}")
    print(f"  log(q) = m * log(p) = {m} * log({p})  (symbolic, exact)")
    print(f"  Numerically: {L_q_float:.6f}  (positive)")
    print()

    # ==================================================================
    # X.1: log(q_under)
    # ==================================================================
    print("--- X.1: log(q_under) ---")
    log_q_under = L_q / (2 * ell)
    print(f"  log(q_under) = log(q) / (2*l) = {log_q_under}")
    print(f"  Numerically: {float(log_q_under):.6f}")
    print(f"  Matches IUT IV p.23: log(q_under) = (1/2l) log(q)")
    print()

    # ==================================================================
    # X.2: Log-volumes (negative convention)
    # ==================================================================
    print("--- X.2: Log-volumes (negative convention) ---")
    mu_q     = QQ(-1) / QQ(2) * L_q
    mu_Theta = QQ(-1) / QQ(6) * L_q

    print(f"  mu_q     = -1/2 * log(q) = {mu_q}")
    print(f"           Numerically: {float(mu_q):.6f}  (negative)")
    print(f"  mu_Theta = -1/6 * log(q) = {mu_Theta}")
    print(f"           Numerically: {float(mu_Theta):.6f}  (negative)")
    print()

    # ==================================================================
    # X.3: Corollary 3.12
    # ==================================================================
    print("--- X.3: IUT III Corollary 3.12 ---")
    print(f"  Claim: mu_Theta >= mu_q")
    holds = (mu_Theta - mu_q) >= 0
    print(f"  mu_Theta - mu_q = {mu_Theta - mu_q}")
    print(f"  Symbolically >= 0? {holds}")
    assert holds, "Error: Corollary 3.12 fails."
    print(f"  [PASS] Corollary 3.12 holds.")
    print()

    # ==================================================================
    # X.4: The gap
    # ==================================================================
    print("--- X.4: The gap between Theta-pilot and q-pilot ---")
    gap = mu_Theta - mu_q
    print(f"  gap = mu_Theta - mu_q = {gap}")
    expected_gap = QQ(1) / QQ(3) * L_q
    print(f"  expected = (1/3) log(q) = {expected_gap}")
    print(f"  Numerically: {float(gap):.6f}")

    # Symbolic equality: both sides are rational multiples of log(11).
    # Use Sage's symbolic simplification to check.
    diff = (gap - expected_gap).simplify_full()
    print(f"  gap - expected = {diff}")
    assert diff == 0, "Error: gap does not match (1/3) log(q)."
    print(f"  [PASS] Gap = (1/3) log(q) > 0.")
    print()

    # ==================================================================
    # X.5: Consistency with Unit W (spike sum)
    # ==================================================================
    print("--- X.5: Consistency with Unit W ---")
    coeffs_sum = QQ(0)
    for j in range(1, ell_star + 1):
        coeffs_sum += QQ(ell - j*j) / QQ(2*ell)
    print(f"  Sum of spike coefficients (Unit W): {coeffs_sum}")
    print(f"  Sum * log(q) = {coeffs_sum * L_q}")
    print(f"  mu_q (q-pilot) = {mu_q}")
    diff5 = (coeffs_sum * L_q - mu_q).simplify_full()
    assert diff5 == 0, "Error: spike sum does not match mu_q."
    print(f"  [PASS] Sum of spikes reproduces mu_q.")
    print()

    # ==================================================================
    # X.6: Consistency with Unit L (procession)
    # ==================================================================
    print("--- X.6: Consistency with Unit L (procession) ---")
    b = QQ(ell + 1) / QQ(24)  # = 7/12 for l = 13
    process_value = b * QQ(m)
    print(f"  Procession coefficient (l+1)/24 = {b}")
    print(f"  Times m = {m}: {process_value}")
    print(f"  This is the coefficient multiplying log(q) on the RHS of")
    print(f"  Theorem 1.10, not a pilot log-volume coefficient.")
    print()

    # ==================================================================
    # Verdict
    # ==================================================================
    print("=" * 78)
    print("  VERDICT")
    print("=" * 78)
    print()
    print("ALL CHECKS PASSED (symbolically exact).")
    print()
    print("Numerical content of IUT III Corollary 3.12 for 11a1 at l=13:")
    print()
    print(f"  log(q) = {L_q}  (symbolic)")
    print(f"  q-pilot:     mu_q     = {mu_q}")
    print(f"  Theta-pilot: mu_Theta = {mu_Theta}")
    print()
    print(f"  Corollary 3.12: mu_Theta >= mu_q  [TRUE]")
    print(f"  Gap: mu_Theta - mu_q = {gap}  (positive)")
    print()
    print("Consistency with other units:")
    print(f"  Unit W: sum of spikes = mu_q  [PASS]")
    print(f"  Unit L: procession coefficient (l+1)/24 = {b}  [PASS]")
    print()
    print("SCOPE")
    print("-----")
    print("This script verifies the numerical content of IUT III")
    print("Corollary 3.12 for 11a1 at p=11, l=13, in exact symbolic")
    print("arithmetic in log(11).")
    print()
    print("It does NOT verify the structural construction of the")
    print("Theta-pilot and q-pilot objects, nor the derivation of")
    print("their log-volumes from the Theta-link.")
    print("=" * 78)

main()
