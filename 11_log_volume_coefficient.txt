# 11_log_volume_coefficient.sage
#
# Unit K of "Independent Exact Verification of the Finite-Sum and
# Scalar-Rearrangement Identities Surrounding IUT IV".
#
# Verifies IUT IV Proposition 1.4(ii):
#   mu_log(log_p(R_i^×)) = -(1/e_i + m_i/(e_i f_i)) * log(p)
#
# where m_i is the order of the p-primary component of the torsion
# subgroup of O_{K_i}^×, defined by |R_i^μ| = p^{m_i} (p^{f_i} - 1).
#
# Classification: Exact (rational arithmetic + field invariants).
# Runs in SageMath 9.5+ or 10.x. Calls main() unconditionally.

from sage.all import *

def main():
    print("=" * 78)
    print("  UNIT K: LOG-VOLUME COEFFICIENT (IUT IV PROP 1.4(ii))")
    print("=" * 78)
    print()
    print("  Formula: mu_log(log_p(R_i^×)) = -(1/e_i + m_i/(e_i f_i)) log(p)")
    print()

    p = 11
    prec = 50

    # ==================================================================
    # Case 1: K_1 = Q_11 (unramified)
    # ==================================================================
    K1 = Qp(p, prec=prec)
    e1 = K1.e()
    f1 = K1.f()

    torsion_order_1 = p - 1
    m1 = 0
    assert p**m1 * (p**f1 - 1) == torsion_order_1, "Mismatch in case 1."

    coeff1 = -(QQ(1)/QQ(e1) + QQ(m1)/(QQ(e1)*QQ(f1)))
    print(f"--- Case 1: K_1 = Q_{p} ---")
    print(f"  e = {e1}, f = {f1}")
    print(f"  Torsion order: {torsion_order_1} (= p^{m1} (p^{f1} - 1))")
    print(f"  m_1 = {m1}")
    print(f"  Coefficient = {coeff1}")
    print(f"  Expected: -1")
    assert coeff1 == QQ(-1), "Case 1 failed."
    print("  [PASS]")
    print()

    # ==================================================================
    # Case 2: K_2 = Q_11(sqrt 11) (tame ramified, no p-torsion)
    # ==================================================================
    R2 = PolynomialRing(Qp(p, prec=prec), 'x2')
    K2 = Qp(p, prec=prec).extension(R2.gen()**2 - p, names='sqrt11')
    e2 = K2.e()
    f2 = K2.f()

    torsion_order_2 = p - 1
    m2 = 0
    assert p**m2 * (p**f2 - 1) == torsion_order_2, "Mismatch in case 2."

    coeff2 = -(QQ(1)/QQ(e2) + QQ(m2)/(QQ(e2)*QQ(f2)))
    print(f"--- Case 2: K_2 = Q_{p}(sqrt {p}) ---")
    print(f"  e = {e2}, f = {f2}")
    print(f"  Torsion order: {torsion_order_2} (= p^{m2} (p^{f2} - 1))")
    print(f"  m_2 = {m2}")
    print(f"  Coefficient = {coeff2}")
    print(f"  Expected: -1/2")
    assert coeff2 == QQ(-1)/QQ(2), "Case 2 failed."
    print("  [PASS]")
    print()

    # ==================================================================
    # Case 3: K_3 = Q_11(zeta_11) (tame ramified, with p-torsion)
    #
    # We use Phi_11(x+1) = ((x+1)^11 - 1)/x, which is Eisenstein at 11.
    # If y is a root, then zeta = y + 1 is a primitive 11th root of unity.
    # ==================================================================
    R3 = PolynomialRing(Qp(p, prec=prec), 'y3')
    y3 = R3.gen()
    eisenstein_poly = sum(binomial(p, k+1) * y3**k for k in range(p))
    K3 = Qp(p, prec=prec).extension(eisenstein_poly, names='yroot')
    e3 = K3.e()
    f3 = K3.f()

    # Q_11(zeta_11) has torsion of order p*(p-1) = 110.
    torsion_order_3 = p * (p - 1)
    m3 = 1
    assert p**m3 * (p**f3 - 1) == torsion_order_3, \
        f"Case 3 mismatch: {p**m3 * (p**f3 - 1)} != {torsion_order_3}"

    coeff3 = -(QQ(1)/QQ(e3) + QQ(m3)/(QQ(e3)*QQ(f3)))
    print(f"--- Case 3: K_3 = Q_{p}(zeta_{p}) ---")
    print(f"  Constructed via Phi_{p}(y+1), root y generates zeta_{p} = y+1.")
    print(f"  e = {e3}, f = {f3}")
    print(f"  Torsion order: {torsion_order_3} (= p^{m3} (p^{f3} - 1))")
    print(f"  m_3 = {m3}")
    print(f"  Coefficient = {coeff3}")
    print(f"  Expected: -1/5")
    assert coeff3 == QQ(-1)/QQ(5), "Case 3 failed."
    print("  [PASS]")
    print()

    # ==================================================================
    # Summary
    # ==================================================================
    print("=" * 78)
    print("  VERDICT")
    print("=" * 78)
    print()
    print(f"  {'Field':>18} | {'e':>3} | {'f':>3} | {'m':>3} | {'Coefficient':>14} | {'Expected':>10}")
    print("-" * 66)
    print(f"  {'Q_11':>18} | {e1:>3} | {f1:>3} | {m1:>3} | {str(coeff1):>14} | {'-1':>10}")
    print(f"  {'Q_11(sqrt 11)':>18} | {e2:>3} | {f2:>3} | {m2:>3} | {str(coeff2):>14} | {'-1/2':>10}")
    print(f"  {'Q_11(zeta_11)':>18} | {e3:>3} | {f3:>3} | {m3:>3} | {str(coeff3):>14} | {'-1/5':>10}")
    print()
    print("ALL CHECKS PASSED.")
    print()
    print("SCOPE")
    print("-----")
    print("This script verifies the exact log-volume coefficient formula of")
    print("IUT IV Proposition 1.4(ii) for three explicit local fields.")
    print()
    print("It confirms:")
    print("  - K_1 = Q_11: coefficient -1 (unramified, no p-torsion)")
    print("  - K_2 = Q_11(sqrt 11): coefficient -1/2 (ramified, no p-torsion)")
    print("  - K_3 = Q_11(zeta_11): coefficient -1/5 (ramified with p-torsion)")
    print()
    print("It does NOT verify the structural log-volume theory of IUT IV")
    print("beyond the coefficient computation.")
    print("=" * 78)

main()
